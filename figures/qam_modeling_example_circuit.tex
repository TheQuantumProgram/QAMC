% Match gate and meter frames. Meter geometry follows Qiskit's mpl renderer:
% a semicircle 70% of the box width and a plain 45-degree needle.
\tikzset{
 thickness/.style={line width=.6pt},
 qam gate box/.style={draw,fill=white,rectangle,minimum size=5.6mm,
   inner sep=0pt,thickness},
 operator/.style={qam gate box,align=center},
 meter/.style={
  qam gate box,
  path picture={
    \draw[line width=.8pt]
      ([xshift=-1.96mm,yshift=-.84mm]path picture bounding box.center)
      arc[start angle=180,end angle=0,radius=1.96mm];
    \draw[line width=.8pt]
      ([yshift=-.84mm]path picture bounding box.center)
      -- ([xshift=1.96mm,yshift=1.12mm]path picture bounding box.center);
  }
}}
